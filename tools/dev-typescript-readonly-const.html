TS Readonly vs Const

Just const

const user = {
  name: "Alice"
};

// ALLOWED! Object mutated.
user.name = "Bob"; 

// ERROR: Reassignment
user = {}; ❌

const + as const (Readonly)

const user = {
  name: "Alice"
} as const;

// ERROR: Property is read-only
user.name = "Bob"; ❌

// ERROR: Reassignment
user = {}; ❌
